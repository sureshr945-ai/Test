import java.util.*;

class Solution {
    private static final int NIL = 0;
    private static final int INF = Integer.MAX_VALUE;
    
    public Boolean solution(int[] A, int[] B, int S) {
        int N = A.length;
        if (N > S) return false;

        // Build adjacency list (1-indexed for Hopcroft-Karp)
        List<List<Integer>> graph = new ArrayList<>();
        graph.add(new ArrayList<>()); // dummy for index 0
        for (int i = 0; i < N; i++) {
            List<Integer> prefs = new ArrayList<>();
            prefs.add(A[i]);
            prefs.add(B[i]);
            graph.add(prefs);
        }

        return hopcroftKarp(graph, N, S) == N;
    }

    private int hopcroftKarp(List<List<Integer>> graph, int N, int S) {
        int[] pairU = new int[N + 1]; // pairU[u] = slot matched to patient u
        int[] pairV = new int[S + 1]; // pairV[v] = patient matched to slot v
        int[] dist = new int[N + 1];
        
        int matching = 0;
        
        while (bfs(graph, pairU, pairV, dist, N)) {
            for (int u = 1; u <= N; u++) {
                if (pairU[u] == NIL && dfsHK(u, graph, pairU, pairV, dist)) {
                    matching++;
                }
            }
        }
        
        return matching;
    }
    
    private boolean bfs(List<List<Integer>> graph, int[] pairU, int[] pairV, 
                        int[] dist, int N) {
        Queue<Integer> queue = new LinkedList<>();
        
        for (int u = 1; u <= N; u++) {
            if (pairU[u] == NIL) {
                dist[u] = 0;
                queue.offer(u);
            } else {
                dist[u] = INF;
            }
        }
        
        dist[NIL] = INF;
        
        while (!queue.isEmpty()) {
            int u = queue.poll();
            if (dist[u] < dist[NIL]) {
                for (int v : graph.get(u)) {
                    if (dist[pairV[v]] == INF) {
                        dist[pairV[v]] = dist[u] + 1;
                        queue.offer(pairV[v]);
                    }
                }
            }
        }
        
        return dist[NIL] != INF;
    }
    
    private boolean dfsHK(int u, List<List<Integer>> graph, int[] pairU, 
                          int[] pairV, int[] dist) {
        if (u != NIL) {
            for (int v : graph.get(u)) {
                if (dist[pairV[v]] == dist[u] + 1) {
                    if (dfsHK(pairV[v], graph, pairU, pairV, dist)) {
                        pairV[v] = u;
                        pairU[u] = v;
                        return true;
                    }
                }
            }
            dist[u] = INF;
            return false;
        }
        return true;
    }
}
