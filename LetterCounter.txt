public class LetterCounter {
    
    public static int solution1_TwoPassIndexTracking(String letters) {
        if (letters == null || letters.isEmpty()) {
            return 0;
        }
        
        int[] lastLowercase = new int[26];
        int[] firstUppercase = new int[26];
        
        // Initialize arrays
        for (int i = 0; i < 26; i++) {
            lastLowercase[i] = -1;
            firstUppercase[i] = Integer.MAX_VALUE;
        }
        
        // First pass: record last lowercase and first uppercase positions
        for (int i = 0; i < letters.length(); i++) {
            char c = letters.charAt(i);
            if (Character.isLowerCase(c)) {
                lastLowercase[c - 'a'] = i;
            } else {
                int index = c - 'A';
                firstUppercase[index] = Math.min(firstUppercase[index], i);
            }
        }
        
        // Count letters meeting the condition
        int count = 0;
        for (int i = 0; i < 26; i++) {
            if (lastLowercase[i] != -1 && 
                firstUppercase[i] != Integer.MAX_VALUE && 
                lastLowercase[i] < firstUppercase[i]) {
                count++;
            }
        }
        
        return count;
    }
}
